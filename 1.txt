// 渲染单词列表
function renderList(list){
  wordList.innerHTML = '';
  if(list.length === 0){
    wordList.innerHTML = '<div class="empty">未匹配到相关单词</div>';
    return;
  }
  list.forEach(item=>{
    const card = document.createElement('div');
    card.className = 'word-card';
    card.innerHTML = `<div class="en">${item.en}</div><div class="cn">${item.cn}</div>`;
    wordList.appendChild(card);
  })
}

// 初始化展示全部单词

searchInput.addEventListener('input',()=>{
  const key = searchInput.value.trim().toLowerCase();
  if(!key){
    renderList(wordData);
    return;
  }
  const filter = wordData.filter(w=>{
    return w.en.toLowerCase().includes(key) || w.cn.includes(key);
  })
  renderList(filter);
})
